// Prueba integral: combina todos los elementos del lenguaje.
// Calcula: suma 1..N, factorial N, busqueda de maximo, fibonacci iterativo.
program main{
	var n, i : int;
	var suma, producto : int;
	var fib, fibPrev, fibActual, aux : int;
	var max, min : int;
	var a, b, c : int;
	var condicion : bool;
	var mensaje : string;

	// retorna el cuadrado de x
	proc cuadrado(x: int) {
		return x * x;
	}

	// retorna el maximo de dos valores
	proc mayor(a: int, b: int) {
		if (a >= b) then { return a; } else { return b; }
	}

	// suma de 1 a n usando while (escribe resultado en variable global suma)
	proc sumaHasta(n: int) {
		suma := 0;
		i := 1;
		while (i <= n) do
		{
			suma := suma + i;
			i++;
		}
	}

	begin;
		write("=== Programa Integral ===");

		// ---------- tipos de datos ----------
		write("-- Tipos --");
		n := 10;
		condicion := true;
		mensaje := "activo";
		write(n);
		write(condicion);
		write(mensaje);

		// ---------- operadores aritmeticos ----------
		write("-- Aritmeticos --");
		a := 15;
		b := 4;
		write("15 + 4 ="); write(a + b);
		write("15 - 4 ="); write(a - b);
		write("15 * 4 ="); write(a * b);
		write("15 % 4 ="); write(a % b);
		write("cuadrado(7) ="); write(cuadrado(7));

		// ---------- operadores relacionales ----------
		write("-- Relacionales --");
		if (a > b) then  { write("15 > 4 : verdadero"); }
		if (a < b) then  { write("error"); } else { write("15 < 4 : falso"); }
		if (a >= 15) then { write("15 >= 15 : verdadero"); }
		if (a == b) then { write("error"); } else { write("15 == 4 : falso"); }
		if (a != b) then { write("15 != 4 : verdadero"); }

		// ---------- operadores logicos ----------
		write("-- Logicos --");
		if ((a > 10) and (b < 10)) then { write("and(V,V) = verdadero"); }
		if ((a < 0) or (b > 0)) then   { write("or(F,V) = verdadero"); }
		if (not (a == 0)) then         { write("not(F) = verdadero"); }

		// ---------- asignaciones ----------
		write("-- Asignaciones acumuladas --");
		c := 0;
		c := c + a;
		c := c + b;
		c := c * 2;
		write("(15+4)*2 ="); write(c);

		// ---------- estatuto de control (if anidado) ----------
		write("-- Control if anidado --");
		a := 8;
		b := 12;
		c := 10;
		if (a > b) then
		{
			write("a es mayor");
		} else
		{
			if (b > c) then
			{
				write("b es mayor que c: verdadero");
			} else
			{
				write("c es mayor o igual a b");
			}
		}

		// ---------- estatuto de repeticion: while ----------
		write("-- Suma 1..10 con while --");
		sumaHasta(10);
		write("suma ="); write(suma);

		// ---------- estatuto de repeticion: for ----------
		write("-- Factorial 6 con for --");
		n := 6;
		producto := 1;
		for (i := 1; i <= n; i++) {
			producto := producto * i;
		}
		write("6! ="); write(producto);

		// ---------- variables dimensionadas simuladas ----------
		write("-- Vector 5 elementos --");
		a := 3; b := 7; c := 1;
		aux := 9; n := 5;
		// mayor entre los 5
		max := mayor(mayor(a, b), mayor(c, mayor(aux, n)));
		write("max(3,7,1,9,5) ="); write(max);

		// ---------- llamadas a funcion ----------
		write("-- Funciones anidadas --");
		write("mayor(suma(2,3), cuadrado(2)) =");
		write(mayor(suma + 5, cuadrado(2)));

		// ---------- fibonacci iterativo ----------
		write("-- Fibonacci iterativo (10 terminos) --");
		fibPrev := 0;
		fibActual := 1;
		i := 2;
		write(fibPrev);
		write(fibActual);
		while (i < 10) do
		{
			fib := fibPrev + fibActual;
			fibPrev := fibActual;
			fibActual := fib;
			write(fib);
			i++;
		}

		write("=== Fin ===");
	end;
}
