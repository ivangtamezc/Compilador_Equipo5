program main{
	var resultado : int;
	var resf : float;

	// procedimiento sin parametros
	proc saludar() {
		write("Hola desde procedimiento!");
	}

	// suma de dos enteros
	proc suma(x: int, y: int) {
		return x + y;
	}

	begin;
		write("Llamadas a funciones");

		// llamada sin retorno capturado (void)
		saludar();

		// llamada con retorno
		resultado := suma(3, 4);
		write("suma(3, 4) =");
		write(resultado);
	end;
}
