// Variables dimensionadas simuladas con variables individuales del mismo tipo.
// La gramatica actual no soporta arreglos con indice, por lo que se declaran
// multiples variables del mismo tipo en un solo var para representar un vector.
program main{
	// vector de 5 elementos (declaracion agrupada)
	var v1, v2, v3, v4, v5 : int;
	var suma, max, min : int;
	var i, n : int;
	begin;
		write("=== Variables Dimensionadas (vector simulado) ===");

		// inicializacion del vector de 5 elementos
		v1 := 12;
		v2 := 45;
		v3 := 7;
		v4 := 30;
		v5 := 21;

		write("Vector:");
		write(v1);
		write(v2);
		write(v3);
		write(v4);
		write(v5);

		// suma de todos los elementos
		suma := v1 + v2 + v3 + v4 + v5;
		write("Suma del vector:");
		write(suma);

		// maximo del vector (busqueda lineal simulada)
		max := v1;
		if (v2 > max) then { max := v2; }
		if (v3 > max) then { max := v3; }
		if (v4 > max) then { max := v4; }
		if (v5 > max) then { max := v5; }
		write("Maximo del vector:");
		write(max);

		// minimo del vector
		min := v1;
		if (v2 < min) then { min := v2; }
		if (v3 < min) then { min := v3; }
		if (v4 < min) then { min := v4; }
		if (v5 < min) then { min := v5; }
		write("Minimo del vector:");
		write(min);

		// promedio (entero)
		write("Promedio (entero):");
		write(suma / 5);

		// segundo vector: float
		write("=== Vector flotante ===");
	end;
}
